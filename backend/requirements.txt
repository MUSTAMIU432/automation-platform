# JWT access-token encode/decode only (identity/tokens.py). Chosen over
# djangorestframework-simplejwt because the project doesn't use DRF and
# simplejwt is built around DRF's request/view/serializer stack - pulling
# it in would mean adopting a chunk of an unrelated framework just for
# token math. PyJWT is a small, framework-agnostic, actively maintained
# library that does only the encode/decode/verify we actually need; the
# refresh-token/session logic it doesn't provide (rotation, revocation,
# server-side storage) is deliberately hand-rolled in identity/models.py
# and identity/authentication.py instead of trusted to a token library.
PyJWT==2.15.0
